这个控制器做了 4 件关键事情：

1. 强制方向连续（dot product 防翻转）
2. 极区保护（避免 forward 和 up 共线）
3. 正交坐标系重建（稳定姿态）
4. quaternion 平滑插值（不会抖）
🧩 直接可用代码（核心类）

你可以把这个当“模块”用👇

class StableFlightController {
  constructor(curve, radius) {
    this.curve = curve
    this.radius = radius

    this.t = 0
    this.speed = 0.0008
    this.direction = 1

    // 上一帧状态（关键）
    this.prevForward = new THREE.Vector3(0, 0, 1)
    this.prevUp = new THREE.Vector3(0, 1, 0)
    this.prevRight = new THREE.Vector3(1, 0, 0)

    this.tmpVec = new THREE.Vector3()
    this.matrix = new THREE.Matrix4()
    this.quaternion = new THREE.Quaternion()
  }

  update(plane) {
    // === 1. 更新时间（支持往返）===
    this.t += this.speed * this.direction

    if (this.t >= 1 || this.t <= 0) {
      this.direction *= -1
      this.t = THREE.MathUtils.clamp(this.t, 0, 1)
    }

    // === 2. 获取位置 ===
    const position = this.curve.getPointAt(this.t)
    const normal = position.clone().normalize()

    // === 3. 高度（贴地飞）===
    const height = 0.2
    const finalPos = normal.clone().multiplyScalar(this.radius + height)

    // === 4. forward（切线）===
    let forward = this.curve.getTangentAt(this.t).normalize()

    // 🔥 关键：方向连续性修复
    if (forward.dot(this.prevForward) < 0) {
      forward.negate()
    }

    // === 5. up（地球法线）===
    let up = normal.clone()

    if (up.dot(this.prevUp) < 0) {
      up.negate()
    }

    // === 6. 极区保护（防止共线）===
    if (Math.abs(forward.dot(up)) > 0.98) {
      // 使用上一帧 right 保持稳定
      const right = this.prevRight.clone()
      forward = new THREE.Vector3().crossVectors(up, right).normalize()
    }

    // === 7. 计算 right ===
    let right = new THREE.Vector3().crossVectors(forward, up).normalize()

    // === 8. 正交化 up（非常关键）===
    up = new THREE.Vector3().crossVectors(right, forward).normalize()

    // === 9. 构建旋转矩阵 ===
    this.matrix.makeBasis(right, up, forward)

    // === 10. 转 quaternion ===
    this.quaternion.setFromRotationMatrix(this.matrix)

    // === 11. 平滑旋转（防抖）===
    plane.quaternion.slerp(this.quaternion, 0.15)

    // === 12. 更新位置 ===
    plane.position.copy(finalPos)

    // === 13. 保存状态（关键）===
    this.prevForward.copy(forward)
    this.prevUp.copy(up)
    this.prevRight.copy(right)
  }
}
🚀 使用方法（很简单）

在你的 main.js：

1️⃣ 创建航线
const start = latLngToVector3(49.28, -123.12, 5)
const end = latLngToVector3(31.23, 121.47, 5)

const curve = new THREE.CatmullRomCurve3([
  start,
  start.clone().multiplyScalar(1.2),
  end.clone().multiplyScalar(1.2),
  end
])
2️⃣ 创建控制器
const controller = new StableFlightController(curve, 5)
3️⃣ 在动画里调用
function animate() {
  requestAnimationFrame(animate)

  if (plane) {
    controller.update(plane)
  }

  renderer.render(scene, camera)
}
